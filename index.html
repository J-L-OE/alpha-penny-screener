import numpy as np
import pandas as pd

# 1. Rolling Anchor-VWAP & Volume Profile (VPOC / VAH Liquiditätszentren)
def calculate_anchored_vwap(df: pd.DataFrame, anchor_idx: int) -> pd.Series:
    df_slice = df.iloc[anchor_idx:].copy()
    typical_price = (df_slice['High'] + df_slice['Low'] + df_slice['Close']) / 3.0
    cum_vol_price = (typical_price * df_slice['Volume']).cumsum()
    cum_vol = df_slice['Volume'].cumsum()
    return cum_vol_price / cum_vol

# 2. Die Golden-Pocket-Formel (Struktureller Retest 0.618 - 0.786 + 0.886 Stop)
def get_fibonacci_entry_zone(p_low: float, p_high: float) -> dict:
    diff = p_high - p_low
    entry_618 = p_high - (0.618 * diff)
    entry_786 = p_high - (0.786 * diff)
    stop_loss = p_high - (0.886 * diff)
    return {
        "zone_high": round(entry_618, 4),
        "zone_low": round(entry_786, 4),
        "invalid_level": round(stop_loss, 4),
    }

# 3. Kompressions-Scan: Bollinger (20, 2.0) in Keltner Channels (20, 1.5 * ATR)
def check_volatility_squeeze(df: pd.DataFrame, period: int = 20, bb_mult: float = 2.0, kc_mult: float = 1.5) -> bool:
    sma = df['Close'].rolling(window=period).mean()
    std = df['Close'].rolling(window=period).std()
    bb_upper = sma + (std * bb_mult)
    bb_lower = sma - (std * bb_mult)

    high_low = df['High'] - df['Low']
    high_close = (df['High'] - df['Close'].shift()).abs()
    low_close = (df['Low'] - df['Close'].shift()).abs()
    tr = high_low.combine(high_close, max).combine(low_close, max)
    atr = tr.rolling(window=period).mean()

    kc_upper = sma + (atr * kc_mult)
    kc_lower = sma - (atr * kc_mult)
    is_squeeze = (bb_upper < kc_upper) & (bb_lower > kc_lower)
    return bool(is_squeeze.iloc[-1])

# 4. 100-Punkte Confluence-Scoring (Signal-Trigger ab Score >= 80)
def evaluate_entry_confluence(price: float, sma200: float, p_low: float, p_high: float,
                              avwap: float, pullback_vol: float, spike_vol: float, spread_pct: float) -> dict:
    fib = get_fibonacci_entry_zone(p_low, p_high)
    score = 0
    if price > sma200: score += 25
    if (fib["zone_low"] <= price <= fib["zone_high"]) or abs(price - avwap) / price <= 0.025: score += 30
    if pullback_vol < (0.50 * spike_vol): score += 25
    if spread_pct < 1.5: score += 20
    return {"confluence_score": score, "trigger_alert": score >= 80, **fib}

2. Qualitätsbewertung: Die 4 Säulen der Due Diligence

Da klassische Kennzahlen wie das KGV bei unprofitablen Microcaps selten greifen, liegt der Fokus auf Substanz, Liquidität und Risikominimierung:

A. Cash Runway & Verwässerung (Dilution Check)Überlebenswichtig

  • Cash Runway: Berechne Barmittel ÷ quartalsweisen operativen Cash-Burn. Wenn die Liquidität für weniger als 9–12 Monate reicht, steht eine Notkapitalerhöhung bevor, die den Kurs massiv drückt.
  • Share Count History: Prüfe die Entwicklung der ausstehenden Aktien (Shares Outstanding) über die letzten 2–3 Jahre. Steigt die Aktienanzahl kontinuierlich um mehr als 15–20 % pro Jahr, finanziert das Management das operative Überleben primär über Kapitalerhöhungen auf Kosten der Altaktionäre.
  • SEC Filings (US-Werte): Suche in den EDGAR-Filings nach Formblättern wie S-3, 424B (Shelf Registrations) oder toxischen Wandelanleihen (Convertible Notes). Liegt ein aktives ATM-Programm (At-The-Market Offering) vor, verkauft das Unternehmen bei jedem Kurssprung neue Aktien direkt in den Markt.

B. Float & EigentümerstrukturAngebots-Dynamik

  • Low Float: Ein geringer frei handelbarer Streubesitz (Float < 15–20 Mio. Aktien) sorgt bei Katalysatoren für schnelle Kursschübe, birgt aber auch hohe Abwärtsvolatilität.
  • Insider-Ownership: Halten Gründer und Vorstand selbst nennenswerte Anteile (idealerweise > 10–15 %) oder verkaufen sie bei jedem Uptrend ("Insider Selling")?

C. Echte Katalysatoren vs. "Pump & Dump"News-Forensik

  • Belastbare Ereignisse: Zulassungsentscheidungen (FDA PDUFA-Dates bei Biotech), signifikante Umsatzsprünge durch Lizenzverträge, profitable Großaufträge oder das Erreichen des Break-even.
  • Red Flags: Aggressive Werbe-Newsletter ("Next 1000% Exploding Stock"), unklare Pressemitteilungen ohne konkrete Vertragszahlen, Reverse Mergers mit dubiosen Briefkastenfirmen oder ein Kurs unter 1 $ über Monate ohne klaren Sanierungsplan.

D. Technische VerifikationChart & Volumen

  • Chartstruktur & Liquiditätspools: Achte darauf, ob der Kurs an markanten Unterstützungen (bspw. 200-Tage-Linie, Volume-Weighted Average Price / VWAP) Boden bildet oder ob es sich lediglich um einen "Dead Cat Bounce" im intakten Abwärtstrend handelt.
  • Volume Profile: Liegt substanzielles Handelsvolumen am aktuellen Preislevel (Point of Control / Value Area) oder wurde der Kurs mit Mini-Umsätzen nach oben gezogen?

3. Execution & Handwerkszeug

1. Ausschließlich Limit-Orders

Niemals Bestens/Billigst (Market Orders) bei Pennystocks handeln, um extremes Slippage durch breite Spreads zu vermeiden.

2. Asymmetrisches CRV (≥ 3:1 oder 4:1)

Positioniere dich nur, wenn das Setup mindestens ein Chance-Risiko-Verhältnis (CRV) von 3:1 oder 4:1 ermöglicht.

3. Striktes Positions-Sizing (1–3 %)

Begrenze spekulatives Micro-Cap-Exposure auf maximal 1–3 % deines Gesamtportfolios.

Teil 2: Wie du Pennystocks mit 100 % bis 500 %+ Multi-Bagger-Potenzial vor der Masse findest

Große Gewinne bei Micro-Caps und Pennystocks (100 % bis 500 %+ Kursgewinne) entstehen nicht durch „billige Aktien“, sondern durch eine radikale Diskrepanz zwischen Markterwartung und Realität. Um Multi-Bagger vor der Masse zu entdecken, musst du nach asymmetrischen Katalysatoren und fundamentalen Wendepunkten suchen.

1. Die 4 Kernmuster für explosive Kursschübe

Muster A: Der Turnaround / Break-Even-Wendepunkt

Mechanismus: Ein Unternehmen verbrennt seit Jahren Geld und wird vom Markt mit Insolvenzabschlag bewertet. Wenn die operative Wende gelingt (z. B. Quartalszahlen zeigen erstmals positiven Cashflow oder operativen Gewinn), muss der Markt das Unternehmen von Grund auf neu bewerten.

Worauf du achtest: Steigende Bruttomargen bei stagnierenden oder sinkenden Verwaltungskosten (Operating Leverage). Der Übergang von negativem zu positivem EBITDA reicht oft für eine Verdopplung der Bewertungsmultiplikatoren.

Muster B: Binäre Katalysatoren (Biotech & Rohstoffe)

Mechanismus: Ein einzelnes Event entscheidet über den wirtschaftlichen Wert des Unternehmens.
• Biotech: FDA-Zulassungen, Phase-2/3-Studiendaten (PDUFA-Dates).
• Exploration/Mining: Erste Bohrergebnisse mit hoher Erz-/Lithiumkonzentration oder offizielle Machbarkeitsstudien (Feasibility Studies).

Vorteil & Risiko: Termine sind über Kalender (z. B. BioPharmCatalyst) monatelang im Voraus bekannt. Scheitert Phase 3, fällt die Aktie oft um 70–90 % (daher Run-up-Exit nutzen!).

Muster C: Das "Pure Play"-Thema in einem Hype-Sektor

Mechanismus: Ein Makro-Trend explodiert (z. B. KI-Infrastruktur, Verteidigung/Drohnen, Festkörperbatterien). Große institutionelle Gelder treiben zuerst die Mega-Caps nach oben. Sobald diese überkauft sind, suchen Privatanleger und Momentum-Fonds nach „kleinen, unentdeckten Profiteuren“ im selben Sektor.

Worauf du achtest: Kleine Firmen, die echte Verträge, Partnerschaften oder Patente mit Marktführern vorweisen können – keine reinen Pressemitteilungs-Hülsen.

Muster D: Short Squeeze & Low-Float-Struktur

Mechanismus: Hohe Short-Quote (Short Interest > 15–20 % des Float) kombiniert mit einem winzigen frei handelbaren Volumen (Float < 10–20 Mio. Aktien).

Dynamik: Tritt eine positive Nachricht ein, versuchen Leerverkäufer gleichzeitig ihre Positionen zu schließen, während kaum Verkaufsorders im Orderbuch stehen – der Kurs schießt vertikal nach oben.

3. Institutionelles Tracking ("Smart Money" folgen) & 4. Asymmetrisches Trade-Management

🐋 Spuren institutioneller Käufer (SEC Filings)

  • SEC Form 13D / 13G: Zeigt an, wenn ein professioneller Investor mehr als 5 % der Unternehmensanteile erworben hat. Steigt ein bekannter Micro-Cap-Hedgefonds aktiv ein, ist das ein starkes Kaufsignal.
  • Form 4 (Cluster-Insiderkäufe): Wenn CEO, CFO und Vorstandsmitglieder zeitgleich mit eigenem Geld am offenen Markt Aktien kaufen (nicht durch Boni/Optionen), deutet das auf hohes Vertrauen in anstehende Quartalszahlen oder Verträge hin.
  • Regelmäßiges SEC-Screening: Täglich nach Insider Purchases > 100k USD bei Micro-Caps filtern (direkt im Tab SEC & Smart-Money Radar).

⚖️ Das asymmetrische Trade-Management

  • Gewinne laufen lassen, Risiko deckeln: Wenn eine Position um 50 % bis 100 % gestiegen ist, nimm 30–50 % des Einsatzes heraus (der Trade ist damit risikofrei). Den Rest lässt du mit einem nachgezogenen Trailing Stop weiterlaufen.
  • Katalysator-Exit: Wenn du wegen einer FDA-Entscheidung oder einer Konferenz eingestiegen bist, verkaufe in den "Run-up" vor dem Event ("Buy the rumor, sell the news"), um das binäre Totalverlustrisiko zu umgehen.
  • Kein verbilligtes Nachkaufen: Wenn ein Pennystock fällt und die Kernhypothese bricht, wird nicht nachgekauft – die Position wird strikt liquidiert.

Teil 3: Optimale Einstiegsbereiche programmatisch bestimmen (Liquiditätskonfluenz statt „Chasing“)

Um optimale Einstiegsbereiche programmatisch in dein Radar einzubinden, gibt es keine einzelne magische Formel. Stattdessen berechnet man Zonen mit hoher Liquiditätskonfluenz – also Bereiche, an denen institutionelle Akteure typischerweise Liquidität abgreifen oder Positionen aufbauen. Wer ein Breakout-Radar nutzt, kauft nicht am Spike („Chasing“), sondern erfasst algorithmisch entweder den Pre-Breakout (Kompression) oder den strukturellen Retest nach dem ersten Schub (Pullback).

1. Volume-Profile & Anchor-VWAP

Anchor VWAP (AVWAP): Verankere den VWAP exakt am Tiefpunkt des letzten signifikanten Impulses oder an der Kerze der Katalysator-News. Der Bereich zwischen AVWAP und AVWAP − 1σ dient als dynamische Re-Entry-Zone.

Volume Point of Control (VPOC & VAH): Liegt der Kurs über dem VPOC und fällt auf das obere Ende der Value Area (VAH = Value Area High, 70 % des Volumens) zurück, fungiert dieses Niveau als starker institutioneller Support.

2. Die Golden-Pocket-Formel

Nach einem ersten initialen Kurssprung (Leg 1) korrigieren Micro-Caps fast nie flach. Sie suchen Liquidität im Fibonacci-Bereich zwischen 61,8 % und 78,6 % des vorangegangenen Impulses, um Stop-Losses von Breakout-Tradern abzufischen:

Range = P_high − P_low
Entry_top (61.8%) = P_high − (0.618 × Range)
Entry_bottom (78.6%) = P_high − (0.786 × Range)
Stop_Loss (88.6%) = P_high − (0.886 × Range)

Vorteil: Trocknet das Volumen in dieser Zone aus (Volume Dry-Up), entsteht oft ein asymmetrisches CRV von 4:1 bis 6:1.

3. Bollinger / Keltner Squeeze

Identifiziert Phasen, in denen Smart Money unauffällig akkumuliert und die Volatilität vor dem Ausbruch maximal kontrahiert ist:

BB (20, 2.0σ) vs. KC (20, 1.5 × ATR)
Squeeze On = (BB_Upper < KC_Upper) ∧ (BB_Lower > KC_Lower)

Entry-Trigger: Sobald der Squeeze aktiv ist und das Momentum (MACD-Histogramm oder Close > SMA 20) nach oben dreht, liegt der Einstieg direkt am Ausbruchspunkt mit engem Stop unter dem SMA 20.
